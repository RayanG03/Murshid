وتـهـدف إلــى إكـسـاب الـطلبـة مـهــارات ومعــارف تدعـم مسـريتـهـم
األكــاديـميــة والـمهـنيـة.
المعسكرات التدريبية
برامــج مـكـثفـة قـصيـرة المـدة تـعتمـد على الـتعلـم التطبيقـي والعـمـل
الـجمــاعي، وتـركـــــز على تطويـــر مهـــارات مـحـددة لدى الـطلـبة من خالل
ورش عمـل وتطبيقـات تفاعلية.
المسابقات المهاريـة واإلبداعية
مسـابقات نوعيـة فـي مجـاالت عمليـة وفنيـة متنوعـة تهدف إلـى تنمية
التفـكري اإلبداعـي ومهـارات حل المشـكالت، وتتيح للطلبـة عرض أعمالهم
وتقييمهـا ضمـن إطـار منظم يشـجع على التطوير المسـتمر.
األنشطة الرياضية
بـرامــج وبطـوالت رياضـــية تتيـح للطلبـة ممارســة األنشطـة البدنية في
بيئة جامعية محفزة، وتسهم في تعزيز نمط الحياة الصحي واكتشاف
المواهب الرياضية وتنميتها.
الربامج التطوعية
مبـادرات وأنشـطة تتيـح للطلبـة المشـاركة فـي خدمـة المجتمـع وتنميـة
حــس المســـؤولية لديهــم، مــن خالل إتـاحـــة فــــرص للمشـاركـــــة فـــي
األعــمـــال التطـوعــــية والمبــادرات المجتمعيــة.

الملتقى العلمي
ُيُعـد الملتـــقى العلــــمي منصـة علميـة ُتُعــــــنى بإبـــــراز إبداعــــات الطلبـة البحثيـة ومشاريعـــــــهم
االبتكــــارية فـي مختلـف التخصصـــــــات، حيـث يتيــــــــح لهـم عـرض أبحـــــــــاثهم ومبـــــــادراتهم أمـــــام
لجــــان علميـــــة متخصصـــــة، بما يســـــهم في تنميــــــة مهـــــــــارات البحـــــــث العلــــــمي واالبتكـــــــــــار
وتعزيــــــز ثقــــــــافة اإلنتــــاج المعــــــــــرفي لـدى الطلبــــــــة
معرض ُصُنـع بيــدي
يعـد معــــــرض صنـــــــــع بيــــــدي منصة سنـــوية داعمة إلبـراز المنتجات
اليدويـة والمشــــــــاريع اإلبداعـــــية، تهـــــــدف إلـى تمكيــــــــن الطلبـــــــــة
أصحــــــــاب األفكــار المبتكـــــرة ورواد األعمـــــال من عـرض منتجـــــــــاتهم
وتسويقـــــــــــها فــي بيئــــــة تحــــّفّــــــــــز تحــــويــــــــل اإلبداعــــــــــــات إلــى
مشـــــــاريع ذات أثــــــر واستدامـــــة

المرافق اجلامعية
السكن الجامعي
مسـاحة سـكنية متكاملـة وآمنـة تسـهم فــي دعــــم اســتـقــرار الـطلـبــــة
خــــالل مسريتهـم الدراسـية، مـن خالل تطويـر الـمرافــق، إلـى جانــب أتمــتة
اإلجـــراءات وتحـسني جـودة الخدمـة بمـا يعـزز تجربـة الطلبـة داخـل البيئـة
الجامعيـة.
السكن الجامعي
تـنظيـــم وتـيـسيــــر تنـقـــل الـطلـبــــة بـــــني مرافـــق الجامعـــة وفــــروعها
مــن خــالل تـوفري وسـائل نقــل منظمـة تسـهم فـي دعـــم استقـرارهــم
األكــــاديمي وتعزيــــــز جــودة تجربتهــم الجامعيــة.
الرعاية الصحية
تتـمززّيز جامعـة الملـك عبدالعزيـز بامتالكهـا مستشـفى جامعـي متقـدم
ُيُعـد أحـد الرصوح الطبيـة الرائـدة، يقدم خدمات صحيـة وتعليمية وبحثية
متطـورة، ويمنـح الـطالب تجربـة تعليميـة عمليـة داخـل بيئـة احرتافيـة
متكاملـة، تجمـع بني أحـدث التقنيـات وكفـاءات طبيـة متخصصـة.
المطعم المركزي
تـوفــيــــر ثالث وجـبـــــات يوميًاً، مــتنـوعـــة ومدعومة لطلبة الجامعة، بما
يسهم فــي دعـــم جـــودة الـحيـــاة الـجـامـعـية وتعزيــز الـبيئــة الـمعيشيـة
داخل الحرم الـجــامـعــــي.
