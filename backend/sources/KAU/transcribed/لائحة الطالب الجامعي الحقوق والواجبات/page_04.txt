 حقوق الطالب
ويُقصــد بهــا تلــك الحقــوق 
التــي تكفلهــا أنظمــة الجامعــة 
فــي المجاليــن األكاديمــي وغيــر 
األكاديمـي لتوفيـر بيئـة تعليميـة 
داعمـة تضمـن لـه حيـاة جامعيـة 
مســتقرة وفقــا إلمكاناتهــا. 
واجبات الطالب
ويُقــصــــد بــهـــــا الــواجبــــات 
األكاديميــة وغيـــــر األكــاديمـــية 
التـي يجـب علـى الطالـب اللتـزام 
بهـا تجـاه الجامعة وذلـك لالرتقاء 
بجـــــودة العـمــــل األكــاديمـــي، 
والعمــل علــى توثيــق العالقــــة 
بيـن الطالـب مـن جهـة واألسـتاذ 
الجامعـي والقطاعـات الجامعيـة 
مـن جهـة أخـرى. 
 يمثــل الطالــب الجامعــي أحــد أهــم الركائــز التــي يعــول عليهــا المجتمــع فــي
بنــاء المســتقبل، وتحقيــق التفاعــل مــع المتغيــرات الفكريــة والعلميــة والثقافيــة 
و القتصاديـة و الجتماعيـة، والتكيـف مـع مسـتجداتها ومـن ثـم إرسـاء دعائـم النهضة 
والتنميـة المسـتدامة. وانطالقـا مـن ضـرورة التذكيـر بحقـوق الطالـب وواجباتـه، فقـد 
تـم وضـع صيغـة الئحـة تحـرص على ضمان حقـوق الطالب والتوعيـة بواجباته وتحليه 
بالسـلوكيات اإلسـ اامية والجامعيـة الفضلـى، والعمـل علـى تنميـة شـعوره ب النتمـاء 
لمجتمعـه ووطنـه. وقـد تـم تقسـيم محتـوي الالئحـة إلـى محوريـن أساسـيين يتعلقـان 
بحقـوق وواجبـات الطالـب فـي المجاليـن األكاديمـي وغيـر األكاديمـي، وذلـك علـى
