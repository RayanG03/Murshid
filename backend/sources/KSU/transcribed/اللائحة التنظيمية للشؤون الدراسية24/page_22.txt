ـقدم الـقسم الأكـاديـمي خـطة تـنفيذيـة لـلبرنـامـج المشـترك ضـمن مـتطلبات إقـرار الـبرامـجُي
:الأكاديمية في جامعة الملك سعود للجنة الدائمة للخطط والنظام الدراسي تتضمن
الـمقررات الـتي سـيدرسـها الـطالـب فـي جـامـعة الـملك سـعود وتـلك فـي الـجهة الشـريـكة عـلى.أ
ـساهـم الـمقررات الـتي سـيدرسـها الـطالـب فـي الـجهة الشـريـكة فـي تـحقق نـواتـج تـعلمُأن ت
الـبرنـامـج الـرئيسـي فـي جـامـعة الـملك سـعود بمسـتوى مـكافـئ لـلمقرات الـتي لـن يـدرسـها
الــطالــب فــي جــامــعة الــملك ســعود فــي حــال كــون الــبرنــامــج المشــترك بــنمط الــدرجــة
.المزدوجة
.ب خــطة تــنفيذ مــعايــير الــجودة والاعــتماد الأكــاديــمي لــلبرنــامــج المشــترك بــما يــتوافــق مــع
.متطلبات هيئة تقويم التعليم والتدريب
.ت .شروط القبول وآلية ومعايير المفاضلة
.ث آلـية احـتساب الـوحـدات الـتدريسـية لـلمقررات والـتقديـر فـي السجـل الأكـاديـمي لـلمقررات الـتي
سـيدرسـها الـطالـب فـي الـجهة الشـريـكة بـما يـتناسـب مـع الأنـظمة والـلوائـح ذات الـصلة فـي
.جامعة الملك سعود
.ج خـطة تـسجيل الـطلاب فـي الـبرنـامـج المشـترك وبـقاء الـطالـب ضـمن سـجلات جـامـعة الـملك
.سعود والجهة الشريكة
.ح .خطة تنقل الطلبة بين جامعة الملك سعود والجهة الشريكة
.خ .آلية معالجة الحركات الأكاديمية للطالب في الجهة الشريكة
.د .صيغة وثيقة التخرج للبرنامج المشترك من جامعة الملك سعود والجهة الشريكة
.ذ .صيغة السجل الأكاديمي من جامعة الملك سعود والجهة الشريكة
.ر .آلية معالجة مشاكل الطلبة الأكاديمية بالتنسيق مع الجهة الشريكة
.ز آلــية اســتقطاب الــطلبة والإعــلان عــن الــبرنــامــج مــن قــبل جــامــعة الــملك ســعود والــجهة
.الشريكة
.س الخـطة الـمالـية بـما يـتضمن: الـرسـوم الـدراسـية لـمقررات جـامـعة الـملك سـعود ومـقررات
الــجهة الشــريــكة، وآلــية الــتمويــل أو الــحصول عــلى الــرســوم الــدراســية ومــشاركــتها بــين
.الجامعة والجهة الشريكة
